from machine import Pin, I2C
import ssd1306
import framebuf
import time
i2c = I2C(sda=Pin(4), scl=Pin(5))


HEXS = "${HEX}"
HEXArray = HEXS.split(',')
cols = int(HEXArray[0])
rows = int(HEXArray[1])
pixelSize = int(HEXArray[2])

display = ssd1306.SSD1306_I2C(cols*pixelSize, rows*pixelSize, i2c)

keyframes = []
for i in range(3, len(HEXArray)):
    keyframes.append(HEXArray[i])

display.fill(0)
#display.contrast(1)

def hex_to_array2d(hex_string, rows, cols):
    result = []
    for i in range(rows):
        row = []
        for j in range(cols):
            index = (i * cols + j) * 2
            hex_byte = hex_string[index:index+2]
            byte_value = int(hex_byte, 16)
            row.append(byte_value)
        result.append(row)
    return result

def draw():
    for i, hex in enumerate(keyframes):
        frame = hex_to_array2d(hex,rows,cols)
        for y, row in enumerate(frame):
            for x, value in enumerate(row):
                display.fill_rect(x*pixelSize, y*pixelSize,x*pixelSize+pixelSize,y*pixelSize+pixelSize,value)
        display.show()
        time.sleep_ms(500)

display.show()
while True:
    draw()
